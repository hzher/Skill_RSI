# 离线验收 Codex 原生轨迹

这份夹具检验一个具体问题：第一轮请求后出现的工具调用、工具返回的记忆，以及下一轮请求中的内容，能否连回同一次调用。例中 `memory_lookup` 返回 `unit=ms`。贡献 Codex reader 或升级 CLI 时，可以用同一组输入核对这条链。

夹具采用 `openai/codex@c0cc9fc820f62e4dbdde68214dd6d792d6c8a7ee` 的 trace schema 2。该版本的 writer 把事件与较大的请求、响应、工具结果分开保存，reducer 顺序重放并校验引用。命令读取一个 bundle，默认把输出写到其中的 `state.json`。[原生事件定义](https://github.com/openai/codex/blob/c0cc9fc820f62e4dbdde68214dd6d792d6c8a7ee/codex-rs/rollout-trace/src/raw_event.rs#L26) · [上游测试夹具](https://github.com/openai/codex/blob/c0cc9fc820f62e4dbdde68214dd6d792d6c8a7ee/codex-rs/rollout-trace/src/reducer/test_support.rs#L120) · [CLI 执行入口](https://github.com/openai/codex/blob/c0cc9fc820f62e4dbdde68214dd6d792d6c8a7ee/codex-rs/cli/src/main.rs#L2112)

## 运行夹具

需要 Python 3 和支持 `codex debug trace-reduce` 的本机 CLI。将下面的 Python 代码块保存为临时脚本，运行 `python3 check_codex_trace.py`。它只在临时目录内构造合成数据并运行离线 reducer，退出时清理该目录，无需模型或凭据。

```python
import json
import subprocess
from pathlib import Path
from tempfile import TemporaryDirectory


def write_json(path, value):
    path.write_text(json.dumps(value), encoding="utf-8")


def make_fixture(root):
    (root / "payloads").mkdir()
    thread, turn, call = "thread-test", "turn-test", "recall-call-1"
    write_json(root / "manifest.json", dict(schema_version=2, trace_id="trace-test",
               rollout_id="run-test", root_thread_id=thread, started_at_unix_ms=0,
               raw_event_log="trace.jsonl", payloads_dir="payloads"))
    events = []

    def payload(name, kind, body):
        path = f"payloads/{name}.json"
        write_json(root / path, body)
        return dict(raw_payload_id=name, kind={"type": kind}, path=path)

    def emit(event_type, context=False, **fields):
        seq = len(events) + 1
        events.append(dict(schema_version=2, seq=seq, wall_time_unix_ms=seq,
                           rollout_id="run-test", thread_id=thread if context else None,
                           codex_turn_id=turn if context else None,
                           payload=dict(type=event_type, **fields)))

    def inference(number, inputs, outputs):
        inference_id, response_id = f"inf-{number}", f"resp-{number}"
        req = payload(f"req-{number}", "inference_request", {"input": inputs})
        res = payload(f"res-{number}", "inference_response",
                      {"response_id": response_id, "output_items": outputs})
        emit("inference_started", True, inference_call_id=inference_id, thread_id=thread,
             codex_turn_id=turn, model="fixture-model", provider_name="fixture", request_payload=req)
        emit("inference_completed", True, inference_call_id=inference_id,
             response_id=response_id, response_payload=res)

    user = dict(type="message", id="u1", role="user",
                content=[{"type": "input_text", "text": "Check the saved unit."}])
    action = dict(type="function_call", id="a1", name="memory_lookup", call_id=call,
                  arguments='{"key":"unit"}')
    output = dict(type="function_call_output", id="o1", call_id=call, output="unit=ms")
    final = dict(type="message", id="a2", role="assistant",
                 content=[{"type": "output_text", "text": "Fixture answer: ms."}])

    emit("rollout_started", trace_id="trace-test", root_thread_id=thread)
    emit("thread_started", thread_id=thread, agent_path="/root")
    emit("codex_turn_started", thread_id=thread, codex_turn_id=turn)
    inference(1, [user], [action])
    emit("history_item_accepted", True, item_id=action["id"])
    emit("tool_call_started", True, tool_call_id="dispatch-1", model_visible_call_id=call,
         requester={"type": "model"}, kind={"type": "other", "name": "memory_lookup"},
         summary={"type": "generic", "label": "memory_lookup"},
         invocation_payload=payload("invoke", "tool_invocation", {"key": "unit"}))
    emit("tool_call_ended", True, tool_call_id="dispatch-1", status="completed",
         result_payload=payload("receipt", "tool_result", {"output": output["output"]}))
    inference(2, [user, action, output], [final])
    emit("history_item_accepted", True, item_id=final["id"])
    emit("codex_turn_ended", codex_turn_id=turn, status="completed")
    emit("thread_ended", thread_id=thread, status="completed")
    emit("rollout_ended", status="completed")
    emit("recording_completed")
    return [json.dumps(event) + "\n" for event in events]


def reduce(root, events):
    (root / "trace.jsonl").write_text("".join(events), encoding="utf-8")
    return subprocess.run(["codex", "debug", "trace-reduce", str(root)],
                          text=True, capture_output=True)


with TemporaryDirectory() as directory:
    root = Path(directory)
    events = make_fixture(root)
    state_path = root / "state.json"

    result = reduce(root, events)
    assert result.returncode == 0, result.stderr
    state = json.loads(state_path.read_text())
    first, second = (state["inference_calls"][k] for k in ("inf-1", "inf-2"))
    tool = state["tool_calls"]["dispatch-1"]
    inputs = second["request_item_ids"]
    assert state["recording_complete"] and state["status"] == "completed"
    assert len(inputs) == 3 and first["request_item_ids"] == inputs[:1]
    assert first["response_item_ids"] == tool["model_visible_call_item_ids"] == inputs[1:2]
    assert tool["model_visible_output_item_ids"] == inputs[2:]
    call_item, returned = (state["conversation_items"][key] for key in inputs[1:])
    receipt = json.loads((root / "payloads/receipt.json").read_text())
    assert call_item["kind"] == "function_call" and returned["kind"] == "function_call_output"
    assert call_item["call_id"] == tool["model_visible_call_id"] == returned["call_id"] == "recall-call-1"
    assert returned["produced_by"] == [{"type": "tool", "tool_call_id": "dispatch-1"}]
    assert tool["raw_result_payload_id"] == "receipt"
    assert returned["body"]["parts"][0]["text"] == receipt["output"] == "unit=ms"
    answer = state["conversation_items"][second["response_item_ids"][0]]
    assert answer["body"]["parts"][0]["text"] == "Fixture answer: ms."
    print("完整链通过", inputs, second["response_item_ids"])

    result = reduce(root, events[:-1])
    assert result.returncode == 0, result.stderr
    prefix = json.loads(state_path.read_text())
    assert prefix["status"] == "completed" and prefix["recording_complete"] is False
    print("缺少末尾记录：可回放，recording_complete=false")

    state_path.unlink()  # 区分本次失败与上一次已有的输出
    (root / "payloads/req-2.json").unlink()
    result = reduce(root, events)
    assert result.returncode != 0 and "req-2" in result.stderr, result.stderr
    assert not state_path.exists()
    print("缺少第二轮请求载荷：回放被拒绝")
```

## 如何解读结果

本机以 `codex-cli 0.169.0-alpha.11.1` 复验，三组断言通过。完整轨迹中的首轮输入为 `conversation_item:1`，首轮产生的调用为 `:2`。第二轮输入依次为 `:1 / :2 / :3`，其中 `:3` 是可追到 `dispatch-1` 的工具输出，第二轮回答为 `:4`。这些编号由 reducer 分配。移植测试宜对比对象之间的引用关系，避免依赖特定编号。删除末尾记录后 reducer 返回 0 并报告记录未闭合；删除引用的载荷后返回 1。

这证明 reader 可以在**给定的原生证据**中区别“工具返回了某条记忆”和“后续请求的输入项含有该结果”。夹具的工具、请求和回答都是手工构造的。实际宿主是否主动检索、网络是否真正发送、模型是否据此改变决策、任务是否改善，仍分别需要运行时来源、对应传输回执、后续动作与独立对照。[Codex 接入与真实宿主验收](README.md)
